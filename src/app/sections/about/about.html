@RestController
@RequestMapping("/api/v1")
public class PortfolioController {

    @GetMapping("/about")
    public ResponseEntity<DeveloperProfile> getAboutMe() {
        var profile = new DeveloperProfile(
            "Mateusz Nasewicz",
            "Full Stack Cloud Engineer",
            "I architect and build scalable full-stack applications with a cloud-first mindset. " +
            "Specializing in Java Spring Boot and Angular, I design resilient systems using " +
            "microservices and event-driven architectures. I leverage AWS and Terraform to " +
            "ensure solutions are deployable, secure, and highly available.",
            List.of("Java", "Spring Boot", "Angular", "AWS", "Terraform")
        );

        return ResponseEntity.ok(profile);
    }

    // Immutable data carrier (Java 17+)
    private record DeveloperProfile(
        String name, 
        String title, 
        String bio, 
        List<String> stack
    ) {}
}
